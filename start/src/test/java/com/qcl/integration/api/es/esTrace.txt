@DisplayName("分页查询-边界值测试")
public void testQueryByPage_BoundaryValues() {

    testPagination(0, 10, "pageNum最小值");     // 第一页

    // pageSize边界值
    testPagination(0, -1, "pageSize负数");      // 应该返回错误

    testPagination(0, 1, "pageSize最小值");     // 已测试
    testPagination(0, 1000, "pageSize较大值");  // 性能测试

}
